(헤더1) // row[1]: , 근로지원인명, 생년월일... // B열: 근로지원인명 const birthRaw = String(row[2] || '').trim(); // C열: 생년월일 const leaveHours = safeParseFloat(row[5]); // F열: 휴가(연차) if (name && birthRaw) { const birthDate = safeDate(birthRaw); const matchKey = `${normalizeString(name)}_${normalizeString(birthDate)}`; // 1차 시도: 이름 + 생년월일 완전 일치 let targetStaff = staffMap[matchKey]; // 2차 시도: 이름만으로 찾기 (동명이인이 없을 때만) if (!targetStaff) { const matchedByName = staffList.filter(s => normalizeString(s.name) === normalizeString(name)); if (matchedByName.length === 1) { targetStaff = matchedByName[0]; } } if (targetStaff) { if (!updates[targetStaff.id]) { updates[targetStaff.id] = { newHistory: [...(targetStaff.leaveHistory || [])], staff: targetStaff }; } const history = updates[targetStaff.id].newHistory; // 중복 등록 방지 (동일 날짜, 동일 타입 확인) const existingIdx = history.findIndex(h => h.date === uploadDate && h.type === uploadType); if (existingIdx !== -1) { history[existingIdx] = { ...history[existingIdx], days: leaveHours }; } else if (leaveHours > 0) { history.push({ id: Date.now().toString() + Math.random(), date: uploadDate, days: leaveHours, type: uploadType, reason: '엑셀 일괄등록', createdAt: new Date().toISOString() }); } updateCount++; } else { failCount++; console.log(`매칭 실패 인원: ${name} (${birthDate})`); } } } }); // Firebase 일괄 업데이트 for (const uid of Object.keys(updates)) { const { newHistory, staff } = updates[uid]; const currentInfo = calculateAnnualLeave(staff.joinDate); // 현재 산정 기간 내에 포함된 연차만 합산 const totalUsedHours = newHistory.reduce((acc, curr) => { if (curr.date >= currentInfo.startDate && curr.date <= currentInfo.endDate) { return acc + (safeParseFloat(curr.days) || 0); } return acc; }, 0); const ref = db.collection('users').doc(uid); batch.update(ref, { annualLeaveUsed: totalUsedHours, leaveHistory: newHistory }); } if (updateCount > 0) { await batch.commit(); alert(`✅ 총 ${updateCount}건의 데이터가 반영되었습니다.\n(매칭 실패: ${failCount}건 - 직원명부를 확인하세요.)`); } else { alert(`⚠️ 반영된 데이터가 없습니다.\n1. 직원명부에 직원이 등록되어 있는지\n2. 엑셀의 이름/생년월일이 명부와 일치하는지 확인하세요.`); } if (fileInputRef.current) fileInputRef.current.value = ''; }).catch(err => alert("오류가 발생했습니다: " + err.message));
